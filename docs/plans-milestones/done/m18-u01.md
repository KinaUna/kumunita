1. **`src/Kumunita.Core/Events/Recurrence.cs`** (new file) — the `Recurrence` enum + the `EventRecurrenceRule` struct, exactly as the design doc's §4 / §9 specify:

   ```csharp
   namespace Kumunita.Core.Events;

   /// <summary>
   /// The M18 recurrence type (D2 — the closed rule shape). The author picks one
   /// in the composer (F1); the pure expansion algorithm (C-M18·2) steps by
   /// <c>Interval</c> units of this type.
   /// </summary>
   public enum Recurrence
   {
       /// <summary>No recurrence (the head row's <c>RecurrenceRule</c> is <c>null</c> for a non-recurring event).</summary>
       None,
       /// <summary>Repeats every <c>Interval</c> day(s).</summary>
       Daily,
       /// <summary>Repeats every <c>Interval</c> week(s), on the same day-of-week as the head.</summary>
       Weekly,
       /// <summary>Repeats every <c>Interval</c> month(s), on the same day-of-month as the head (clamped to the month's last day if the head's day is beyond the month's length — e.g. head on the 31st, a 30-day month clamps to the 30th).</summary>
       Monthly,
       /// <summary>Repeats every <c>Interval</c> year(s), on the same month-day as the head (Feb 29 clamps to Feb 28 in a non-leap year).</summary>
       Yearly,
   }

   /// <summary>
   /// The M18 recurrence rule (D2 — the closed rule shape). Carried **only on the
   /// head row** (D3 — the head's <c>RecurrenceRule</c> is non-null; every
   /// non-head occurrence's <c>RecurrenceRule</c> is <c>null</c>). The
   /// <c>Count</c> / <c>Ends</c> fields are mutually exclusive in intent (the
   /// composer disables one when the other is set — D7); if both are set,
   /// <c>Count</c> wins.
   /// </summary>
   public sealed record EventRecurrenceRule
   {
       public Recurrence Recurrence { get; init; }
       /// <summary>The step size (≥ 1, default 1). "Every 2 weeks" is <c>Recurrence.Weekly</c> + <c>Interval: 2</c>.</summary>
       public int Interval { get; init; } = 1;
       /// <summary>Optional: the total number of occurrences including the head. Wins over <c>Ends</c> if both are set.</summary>
       public int? Count { get; init; }
       /// <summary>Optional: the last occurrence's <c>Start</c> must be ≤ <c>Ends</c>.</summary>
       public DateTimeOffset? Ends { get; init; }
   }
   ```
